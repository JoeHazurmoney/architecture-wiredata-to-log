**Goal:** Fix every native-mode bug found while deploying rc3 on a Raspberry Pi 5 with AlmaLinux 10, each with a test, prove the fixes on the Pi, and release them as v0.3.0rc4.

**Architecture:** Small, independent fixes in the deploy scripts, systemd units, RPM spec and one receiver module. A new CI step runs the real scripts as root in network namespaces, because the bugs only showed against a real kernel, nftables and NIC. The SELinux label ships as a CIL module (no new permissions).

**Tech Stack:** bash, Python 3.9+ standard library, pytest, systemd units, nftables, iproute2, RPM spec, SELinux CIL, GitHub Actions.

**Spec:** `docs/superpowers/plans/2026-10-10-alma10-deploy/notes.md` (PR #22, branch `alma10-deploy-notes`): rows A2-2, A2-3, A2-4, A2-5, A6-1, A6-6, A6-7, R-8.

import json

import subprocess

CAPTURE = os.path.join(REPO, "deploy", "receiver", "capture-nic.sh")
EGRESS = os.path.join(REPO, "deploy", "log-server", "egress-nic.sh")

# A stub command: records "name args" (and stdin for `-f -`), then answers from rules.json,
# {"call prefix": [exit code, stdout, stderr]}; the first matching prefix wins, else exit 0.
STUB = '''#!/usr/bin/env python3
import json, os, sys
d = os.environ["STUBS"]
call = " ".join([os.path.basename(sys.argv[0])] + sys.argv[1:])
data = sys.stdin.read() if call.endswith(" -f -") else ""
with open(os.path.join(d, "calls.jsonl"), "a") as f:
    f.write(json.dumps({"call": call, "stdin": data}) + "\\n")
with open(os.path.join(d, "rules.json")) as f:
    rules = json.load(f)
for prefix, (code, out, err) in rules.items():
    if call.startswith(prefix):
        sys.stdout.write(out)
        sys.stderr.write(err)
        sys.exit(code)
'''


def run_script(tmp_path, script, env, rules, names=("ip", "nft", "ethtool", "nmcli", "lldpcli")):
    stubs = tmp_path / "stubs"
    stubs.mkdir()
    for n in names:
        p = stubs / n
        p.write_text(STUB)
        p.chmod(0o755)
    (stubs / "rules.json").write_text(json.dumps(rules))
    r = subprocess.run(["bash", script], universal_newlines=True,
                       stdout=subprocess.PIPE, stderr=subprocess.PIPE,
                       env=dict(os.environ, PATH="%s:%s" % (stubs, os.environ["PATH"]),
                                STUBS=str(stubs), **env))
    log = stubs / "calls.jsonl"
    calls = [json.loads(x) for x in log.read_text().splitlines()] if log.exists() else []
    return r, calls


def capture_env(tmp_path):
    f = tmp_path / "capture.env"
    f.write_text("IFACE=test0\n")
    return {"WD2L_CAPTURE_ENV": str(f)}


def test_capture_nic_keeps_going_when_the_nic_has_no_ring_settings(tmp_path):
    # veth and many USB NICs answer `ethtool -g` with "Operation not supported".

    assert r.returncode == 0, r.stderr
    names = [c["call"] for c in calls]
    assert "ip link set test0 up" in names
    assert not any(c.startswith("ethtool -G") for c in names)
    assert "nft -f -" in names


def test_capture_nic_checks_the_same_ruleset_it_loads(tmp_path):
    r, calls = run_script(tmp_path, CAPTURE, capture_env(tmp_path), {})
    assert r.returncode == 0, r.stderr

    assert [c["call"] for c in nft] == ["nft -c -f -", "nft -f -"]
    assert nft[0]["stdin"] == nft[1]["stdin"]
    text = nft[1]["stdin"]
    assert text.splitlines()[:2] == ["table netdev wd2l_notx", "delete table netdev wd2l_notx"]
    assert 'hook egress device "test0"' in text and "policy drop;" in text

    assert "syntax error" in r.stderr

# `ip` for the watcher: neighbour state lives in a file; `ip -o monitor` flushes it (as a NIC
# down/up does), prints one link event and exits, which ends the watcher's loop.
WATCH_IP = '''#!/bin/sh
echo "$*" >> "$STUBS/calls"
case "$1 $2" in
  "neigh show") cat "$STUBS/neigh" ;;
  "neigh replace") echo "$3 dev $9 lladdr $5 PERMANENT" > "$STUBS/neigh" ;;
  "-o monitor") [ -n "$KEEP" ] || : > "$STUBS/neigh"; echo "4: test0: <BROADCAST,UP,LOWER_UP> state UP" ;;
esac
'''


def run_watch(tmp_path, neigh, keep=""):
    stubs = tmp_path / "stubs"
    stubs.mkdir()
    ip = stubs / "ip"
    ip.write_text(WATCH_IP)
    ip.chmod(0o755)
    (stubs / "neigh").write_text(neigh)
    env = dict(os.environ, PATH="%s:%s" % (stubs, os.environ["PATH"]), STUBS=str(stubs), KEEP=keep,
               **egress_env(tmp_path))
    r = subprocess.run(["bash", WATCH], env=env, universal_newlines=True, timeout=20,
                       stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    calls = (stubs / "calls").read_text().splitlines()
    return r, [c for c in calls if c.startswith("neigh replace")]


def test_arp_watch_puts_the_entry_back_after_a_flush(tmp_path):
    r, replaces = run_watch(tmp_path, neigh="")
    assert r.returncode == 0, r.stderr
    assert replaces == ["neigh replace 10.9.9.2 lladdr 02:00:00:00:00:09 nud permanent dev test0"] * 2

```bash
#!/usr/bin/env bash
# Keep the black hole's static ARP entry on the egress NIC. Taking the NIC down and up (or a
# driver reset) flushes even permanent neighbour entries, and the sender can't resolve the black
# hole by itself: nothing answers, and the egress firewall drops ARP. Nothing would cross until
# egress-nic.sh ran again. Run by wd2l-egress-arp.service: it puts the entry back at start and
# after every link or neighbour change, and logs each time it does.
#
# Settings come from /etc/wd2l/egress.env (WD2L_EGRESS_ENV overrides the path, for tests).
set -euo pipefail
ENV_FILE="${WD2L_EGRESS_ENV:-/etc/wd2l/egress.env}"
[ -f "$ENV_FILE" ] && . "$ENV_FILE"
IFACE="${IFACE:-ens224}"
BH_IP="${BH_IP:-10.255.255.2}"
BH_MAC="${BH_MAC:-02:00:00:00:00:02}"

ensure() {
  if [ -z "$(ip neigh show to "$BH_IP" dev "$IFACE" nud permanent 2>/dev/null)" ] &&
     ip neigh replace "$BH_IP" lladdr "$BH_MAC" nud permanent dev "$IFACE" 2>/dev/null; then
    echo "egress-arp: restored $BH_IP lladdr $BH_MAC on $IFACE"
  fi
}

ensure
# Any link or neighbour event: check again (cheap, and events are rare on this host).
ip -o monitor link neigh | while read -r _; do ensure; done
```

## What changed since rc3

Fixes found deploying rc3 on AlmaLinux 10 (Raspberry Pi 5, SELinux enforcing):

  `egress.env` has a new `PORT` setting (default 5514).

* **The receiver keeps running while its capture NIC is down,** and reports it (rule 1).
* **The native units ask chronyd over UDP,** as the containers do, so time checks no longer log
  an SELinux denial each time.
* **The sender and receiver wait for the clock at boot** (`chrony-wait`), so a reboot no longer
  raises a time alarm.
